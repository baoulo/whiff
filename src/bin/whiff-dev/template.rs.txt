    fn extract(&self, src: &mut dyn Source, cx: &mut Context<'_>) -> Result<(), ExtractError> {
        // The first 64 KiB are pre-read in `src.head()`; use `src.read_vec(offset, len)?`
        // for anything else. `ByteReader` never panics: short input becomes `Truncated`.
        // Text formats: `let text = cx.read_text(src)?.text;` (bounded and decoded),
        // then `cx.emit_text(TextRole::Body, ..)` and `cx.emit_heading(level, ..)`.
        let r = ByteReader::new(src.head());
        // TODO: parse the header and write what you find through `cx`, for example:
        //   let mut r = r.skip(4)?;                 // the magic
        //   let (w, h) = (r.be_u32()?, r.be_u32()?);

        //   cx.content().title(title);
        //   cx.format_set("version", version);    // snake_case keys, unit suffixes
        //   cx.emit_block(BlockKind::XMP, xmp);    // shared decoders handle EXIF/XMP/IPTC/ICC
        let _ = (r, cx);
        Ok(())
    }
}
